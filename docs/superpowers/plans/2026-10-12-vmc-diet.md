## Global Constraints

- One branch per stage off `main`: `vmc-diet-1`, `vmc-diet-2`, `vmc-diet-3`, `vmc-diet-4`. Never commit to `main`. Each stage is one PR; the next stage branches from `main` after the previous PR merged.

- **Behaviour-preserving.** Every log string, its sink (`logger` / `transaction=True` / `vending=True`) and level, every `recorder.record(...)` row, every customer message and its order relative to other side effects, and every `StatusOutputs` call stay exactly as on `main`. ## PR gate (every stage)

1. `gh pr create --base main --head <branch> --title "<title>" --body-file <body>`; Copilot is auto-requested as reviewer on this repo.
2. Run the gate (scratch script `pr_gate.py`, below) until it exits 0: CI checks (`lint`, `test`, `compose-config`) all success **and** a Copilot review exists for the PR's current head commit **and** no unresolved review thread remains.
3. For every unresolved Copilot thread the gate prints: apply the `superpowers:receiving-code-review` discipline. A valid finding is fixed in a new commit; an invalid one gets a reply explaining why, then the thread is resolved (GraphQL `resolveReviewThread`). A push re-runs CI; the gate then waits for a Copilot review of the new head (re-request Copilot once if none arrives within 10 minutes).
4. Merge: `gh pr merge <n> --merge --delete-branch`. Then `git checkout main`, `git pull`.

`pr_gate.py` (scratchpad, not committed): polls every 60 s for up to 40 minutes, using `gh pr view <n> --json headRefOid,statusCheckRollup` and the GraphQL query `pullRequest(number){reviews(last:20){nodes{author{login} state commit{oid}}} reviewThreads(first:50){nodes{id isResolved comments(first:10){nodes{author{login} path line body}}}}}`. Exit 0 when green, Copilot-reviewed at head, and zero unresolved threads; exit 2 with the unresolved threads as JSON; exit 1 on a failed check (print the failing check's name and URL).

    """Captures loguru records with their bound extras and level."""

    def __init__(self):
        self.records: list[tuple[str, str, dict]] = []  # (level, message, extra)
        self._id = logger.add(self._write, level="DEBUG")

    def _write(self, msg):
        r = msg.record
        self.records.append((r["level"].name, r["message"], dict(r["extra"])))

    def close(self):
        logger.remove(self._id)

    def lines(self, *, sink: str | None = None) -> list[tuple[str, str]]:

```python
"""Size ratchet for controller/vmc.py (vmc-diet design §3.6).

Counts, with ast + tokenize, the VMC class's code lines (non-blank,
non-comment, non-docstring), the file's total lines, and the largest
method's code lines. Over a budget fails; more than 10% under a budget
fails too, naming the number to lower it to -- budgets only move down.
"""

import ast
from pathlib import Path

VMC_PATH = Path(__file__).resolve().parent.parent / "controller" / "vmc.py"

# Budgets. Lower them when a stage lands; never raise them.
CLASS_CODE_LINES = 0  # set to the measured value in Step 3
FILE_LINES = 0
LARGEST_METHOD_CODE_LINES = 0


def measure(path: Path = VMC_PATH) -> dict:
    src = path.read_text(encoding="utf-8")
    lines = src.splitlines()
    tree = ast.parse(src)
    cls = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == "VMC")
    doc: set[int] = set()
    for n in ast.walk(cls):
        if isinstance(n, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)) and n.body:
            b = n.body[0]
            if isinstance(b, ast.Expr) and isinstance(b.value, ast.Constant) and isinstance(b.value.value, str):
                doc.update(range(b.lineno, b.end_lineno + 1))

    def is_code(i: int) -> bool:
        s = lines[i - 1].strip()
        return bool(s) and not s.startswith("#") and i not in doc

    methods = {
        n.name: sum(1 for i in range(n.lineno, n.end_lineno + 1) if is_code(i))
        for n in cls.body
        if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))
    }
    largest = max(methods, key=methods.get)
    return {
        "class_code_lines": sum(1 for i in range(cls.lineno, cls.end_lineno + 1) if is_code(i)),
        "file_lines": len(lines),
        "largest_method": largest,
        "largest_method_code_lines": methods[largest],
    }


def _check(name: str, actual: int, budget: int) -> None:
    assert actual <= budget, f"{name}: {actual} over budget {budget}"
    assert actual >= budget * 0.9, f"{name}: {actual} is well under budget; lower {name} to {actual}"


def test_vmc_size_within_budget():
    m = measure()
    _check("CLASS_CODE_LINES", m["class_code_lines"], CLASS_CODE_LINES)
    _check("FILE_LINES", m["file_lines"], FILE_LINES)
    _check(
        f"LARGEST_METHOD_CODE_LINES ({m['largest_method']})",
        m["largest_method_code_lines"],
        LARGEST_METHOD_CODE_LINES,
    )


if __name__ == "__main__":
    print(measure())
```
